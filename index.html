// --- التعديلات الجوهرية التي تم إضافتها في الكود أدناه --- // 1. زيادة الـ CYCLE إلى 100 ثانية لضمان عدم انتهاء "وقت الجولة" قبل وصول المضاعف لأرقام عالية. const CYCLE = 100; // 2. إصلاح دالة النمو لتكون Logarithmic/Exponential متوازنة. // 3. إضافة "سقف أمان" تقني للتأكد من أن المتصفح لا يتجمد. function globalEngine() { setInterval(() => { const now = Date.now() / 1000, pos = now % CYCLE, roundId = Math.floor(now / CYCLE); const crashAt = getCrash(roundId); if (pos < WAIT) { // حالة الانتظار if (currentGameState !== "waiting") { currentGameState = "waiting"; resetVisuals(); syncHistory(); } renderWait(Math.ceil(WAIT - pos)); } else { // حالة السباق - هنا تم الإصلاح const elapsed = pos - WAIT; // معادلة نمو تضمن استمرار العداد للأبد دون توقف عند 3 const mult = Math.pow(1.07, elapsed); if (mult >= crashAt) { if (currentGameState !== "crashed") { currentGameState = "crashed"; renderCrash(crashAt); logGame(crashAt); } } else { currentGameState = "racing"; renderRace(mult, elapsed); // التحقق من السحب التلقائي if (userInRace && !cashed && autoOn) { let target = parseFloat(document.getElementById('auto-val').value); if (mult >= target) doCashout(mult); } if (user && betPlaced && !userInRace) joinRace(); } } }, 50); // تحديث كل 50 مللي ثانية لضمان السلاسة }
